const HL_KEYWORDS = new Set(['сказать','пусть','если','то','иначе','пока','повторить','раз','функция','вернуть','класс','новый','это']);
const HL_BUILTINS = new Set(['добавить_в','удалить_из','размер','ключи_словаря','значения_словаря','получить','отправить','создать_элемент','стилизовать','удалить_элемент','вывести_на_страницу','ввод']);
const HL_FUNCS = new Set(['длина','случайное','корень','степень','округлить','модуль','в_верхний','в_нижний','print', 'спросить_ии', 'ии']);
const HL_BOOLS = new Set(['истина','ложь','ноль']);
const HL_OPS = new Set(['и','или','не']);

// Регулярное выражение для поиска слов, строк и чисел
const HL_TOKEN_RE = /(\/\/[^\n]*)|("(?:[^"\\\n]|\\.)*")|('(?:[^'\\\n]|\\.)*')|(\b\d+(?:\.\d+)?\b)|([а-яёa-z_][а-яёa-z0-9_]*)/gi;

function escapeHtmlRaw(s) {
  return s.replace(/&/g, '&').replace(//g, '>');
}

function updateHighlight() {
  if (!highlight) return;
  let html = escapeHtmlRaw(editor.value);

  html = html.replace(HL_TOKEN_RE, (match, comment, dq, sq, num, word) => {
    if (comment) return '' + comment + '';
    if (dq)      return '' + dq + '';
    if (sq)      return '' + sq + '';
    if (num)     return '' + num + '';
    if (word) {
      const lower = word.toLowerCase();
      if (HL_BOOLS.has(lower))    return '' + word + '';
      if (HL_KEYWORDS.has(lower)) return '' + word + '';
      if (HL_BUILTINS.has(lower)) return '' + word + '';
      if (HL_FUNCS.has(lower))    return '' + word + '';
      if (HL_OPS.has(lower))      return '' + word + '';
      return word;
    }
    return match;
  });

  highlight.innerHTML = html + '\n';
  updateLineNumbers();
}

function replaceOps(s) {
  const start = '(? '  '.repeat(indent);
  const errors = [];

  for (let i = 0; i < lines.length; i++) {
    const raw = lines[i];
    const lineNum = i + 1;
    const line = raw.trim();

    if (!line || line.startsWith('//')) continue;

    try {
      // 1. Закрытие скобки
      if (line === '}') {
        indent = Math.max(0, indent - 1);
        const nextLine = (i + 1 < lines.length) ? lines[i + 1].trim() : '';
        if (!nextLine.startsWith('иначе')) {
          out.push(pad() + '}');
          if (inClass > 0 && indent < inClass) inClass = 0;
        }
        continue;
      }

      // 2. Иначе
      if (line.startsWith('} иначе') || line.startsWith('иначе')) {
        if (line.startsWith('} иначе')) indent = Math.max(0, indent - 1);
        out.push(pad() + '} else {');
        indent++;
        continue;
      }

      // 3. Сказать
      if (line.startsWith('сказать ')) {
        out.push(pad() + 'print(' + replaceOps(line.slice(8).trim()) + ');');
        continue;
      }

      // 4. ИИ (короткая команда)
      if (line.startsWith('ии ') || line.startsWith('спросить_ии ')) {
        const promptText = line.replace(/^(ии|спросить_ии)\s+/, '');
        out.push(pad() + 'print(await спросить_ии(' + replaceOps(promptText) + '));');
        continue;
      }
                  // 5. Ввод (диалог с пользователем)
      if (line.includes(' = ввод ')) {
        const idx = line.indexOf(' = ввод ');
        const left = line.slice(0, idx).trim();
        const promptText = replaceOps(line.slice(idx + 8).trim());
        const varName = left.replace(/^пусть\s+/, '').trim();
        out.push(pad() + (left.startsWith('пусть ') ? 'let ' : '') + varName + ' = await ввод(' + promptText + ');');
        continue;
      }

      // 6. ИИ с сохранением в переменную
      if (line.includes(' = спросить_ии ')) {
        const idx = line.indexOf(' = спросить_ии ');
        const left = line.slice(0, idx).trim();
        const promptText = replaceOps(line.slice(idx + 14).trim());
        const varName = left.replace(/^пусть\s+/, '').trim();
        out.push(pad() + (left.startsWith('пусть ') ? 'let ' : '') + varName + ' = await спросить_ии(' + promptText + ');');
        continue;
      }

      // 7. Объявление переменной (пусть)
      if (line.startsWith('пусть ')) {
        out.push(pad() + 'let ' + replaceOps(line.slice(6).trim()) + ';');
        continue;
      }

      // 8. Условие (если)
      if (line.startsWith('если ') && line.endsWith('{')) {
        let cond = line.slice(5, -1).trim().replace(/\s+то\s*$/i, '');
        out.push(pad() + 'if (' + replaceOps(cond) + ') {');
        indent++;
        continue;
      }

      // 9. Цикл (пока)
      if (line.startsWith('пока ') && line.endsWith('{')) {
        let cond = line.slice(5, -1).trim().replace(/\s+то\s*$/i, '');
        out.push(pad() + 'while (' + replaceOps(cond) + ') {');
        indent++;
        continue;
      }
                  // 10. Цикл (повторить N раз)
      if (line.startsWith('повторить ') && /раз\s*\{$/.test(line)) {
        const n = replaceOps(line.replace(/^повторить\s+/, '').replace(/раз\s*\{$/, '').trim());
        out.push(pad() + 'for (let __i = 0; __i < ' + n + '; __i++) {');
        indent++;
        continue;
      }

      // 11. Объявление класса
      if (line.startsWith('класс ') && line.endsWith('{')) {
        const name = line.slice(6, -1).trim();
        out.push(pad() + 'class ' + name + ' {');
        indent++;
        inClass = indent;
        continue;
      }

      // 12. Объявление функции
      if (line.startsWith('функция ') && line.endsWith('{')) {
        const rest = line.slice(8, -1).trim();
        // Если внутри класса, не пишем function
        out.push(pad() + (inClass > 0 && indent > inClass ? 'async ' : 'async function ') + rest + ' {');
        indent++;
        continue;
      }

      // 13. Возврат значения
      if (line.startsWith('вернуть ')) {
        out.push(pad() + 'return ' + replaceOps(line.slice(8).trim()) + ';');
        continue;
      }

      // 14. Создание объекта (новый)
      if (line.startsWith('новый ')) {
        out.push(pad() + replaceOps(line).replace(/^новый\s+/, 'new ') + ';');
        continue;
      }
                // 15. Работа с массивами
      if (line.startsWith('добавить_в ')) {
        const rest = replaceOps(line.slice(10).trim()).split(/\s+/);
        out.push(pad() + rest[0] + '.push(' + rest.slice(1).join(' ') + ');');
        continue;
      }
      if (line.startsWith('размер ')) {
        out.push(pad() + 'print(Object.keys(' + replaceOps(line.slice(7).trim()) + ').length);');
        continue;
      }

      // 16. HTTP Запросы
      if (line.includes(' = получить ')) {
        const idx = line.indexOf(' = получить ');
        const left = line.slice(0, idx).trim();
        const url = replaceOps(line.slice(idx + 11).trim());
        const varName = left.replace(/^пусть\s+/, '').trim();
        out.push(pad() + (left.startsWith('пусть ') ? 'let ' : '') + varName + ' = await получить(' + url + ');');
        continue;
      }
      if (line.startsWith('отправить ')) {
        out.push(pad() + 'await отправить(' + replaceOps(line.slice(10).trim()) + ');');
        continue;
      }

      // 17. DOM (Работа с HTML)
      if (line.startsWith('создать_элемент ') || line.startsWith('стилизовать ') || line.startsWith('вывести_на_страницу ')) {
        const spaceIdx = line.indexOf(' ');
        out.push(pad() + line.slice(0, spaceIdx) + '(' + replaceOps(line.slice(spaceIdx + 1).trim()) + ');');
        continue;
      }

      // 18. Обычное выражение (если ничего не подошло)
      out.push(pad() + replaceOps(line) + ';');

    } catch (err) {
      errors.push({ lineNum, text: line, msg: err.message });
    }
  }

  // Проверка незакрытых скобок в конце файла
  const balance = checkBalance(source);
  if (balance > 0) errors.push({ lineNum: lines.length, text: '(конец файла)', msg: 'Не закрыто ' + balance + ' скобок "}"' });
  else if (balance < 0) errors.push({ lineNum: lines.length, text: '(конец файла)', msg: 'Лишних скобок: ' + Math.abs(balance) });

  // Если есть ошибки, выбрасываем их
  if (errors.length) {
    throw new Error(errors.map(e => 'Строка ' + e.lineNum + ': ' + e.msg + '\n   >> ' + e.text).join('\n\n'));
  }

  // Возвращаем сгенерированный JS код
  return out.join('\n');
}

function buildBuiltins(logFn) {
  // Создаем контейнер для вывода HTML-элементов (под редактором)
  let domContainer = document.getElementById('bc-dom-output');
  if (!domContainer) {
    domContainer = document.createElement('div');
    domContainer.id = 'bc-dom-output';
    domContainer.style.cssText = 'padding:16px; border-top:1px solid var(--border); background:var(--card); max-height:200px; overflow-y:auto;';
    document.querySelector('.editor-panel').appendChild(domContainer);
  }
  domContainer.innerHTML = ''; // Очищаем при новом запуске

  return {
    // 1. Вывод текста в консоль
    print: (...args) => logFn(args.join(' '), 'ok'),
    
    // 2. Математика и утилиты
    случайное: (min, max) => Math.floor(Math.random() * (max - min + 1)) + min,
    корень: (x) => Math.sqrt(x),
    степень: (x, y) => Math.pow(x, y),
    длина: (x) => (Array.isArray(x) || typeof x === 'string') ? x.length : 0,
    размер: (x) => (x && typeof x === 'object') ? Object.keys(x).length : 0,
    округлить: (x) => Math.round(x),
    модуль: (x) => Math.abs(x),
    в_верхний: (s) => String(s).toUpperCase(),
    в_нижний: (s) => String(s).toLowerCase(),

    // 3. Ввод с клавиатуры
    ввод: (promptText) => window.prompt(promptText || "Введите значение:"),

    // 4. HTTP Запросы
    получить: async (url) => {
      logFn('🌐 GET ' + url, 'info');
      try { 
        return await (await fetch(url)).json(); 
      } catch (err) { 
        logFn('❌ Ошибка запроса: ' + err.message, 'err'); 
        return null; 
      }
    },

    // 5. Работа с HTML (DOM)
    создать_элемент: (type, text, id) => {
      const el = document.createElement(type);
      el.textContent = text || '';
      if (id) el.id = id;
      domContainer.appendChild(el);
    },
    
    // 6. Встроенный ИИ (можно оставить на будущее, если захочешь вернуть)
    спросить_ии: async (promptText) => {
      logFn('🤖 Запрос к ИИ...', 'info');
      try {
        const res = await fetch(PROXY_URL, { 
          method: 'POST', 
          headers: { 'Content-Type': 'application/json' }, 
          body: JSON.stringify({ platform: 'gemini', action: 'generate', payload: { prompt: promptText } }) 
        });
        const data = await res.json();
        if(data.ok) return data.answer;
        else { logFn('❌ Ошибка ИИ: ' + data.error, 'err'); return "Ошибка"; }
      } catch(err) { 
        logFn('❌ Сеть: ' + err.message, 'err'); 
        return "Ошибка сети"; 
      }
    }
  };
}

async function runCode() {
  clearConsole();
  errorLines = []; // Сбрасываем прошлые ошибки
  updateHighlight(); // Убираем красные строки
  
  const code = editor.value.trim();
  if (!code) { 
    logConsole('// Нечего запускать — напиши код!', 'info'); 
    return; 
  }

  let jsCode;
  try {
    jsCode = bcToJs(code); // Переводим BC+ в JS
  } catch (err) {
    logConsole('❌ ОШИБКА ПЕРЕВОДА BC+ → JS', 'err');
    err.message.split('\n').forEach(l => logConsole(l, 'err'));
    
    // Ищем номер строки из текста ошибки, чтобы подсветить её
    const match = err.message.match(/Строка (\d+):/);
    if (match) { 
      errorLines.push(parseInt(match[1])); 
      updateHighlight(); 
    }
    statusEl.textContent = 'Ошибка перевода';
    statusEl.style.color = 'var(--danger)';
    return;
  }

  logConsole('▶ Запуск main.bc+', 'info');
  
  try {
    // Создаем асинхронную функцию, чтобы работали await (ввод, запросы)
    const AsyncFn = Object.getPrototypeOf(async function(){}).constructor;
    const builtinsObj = buildBuiltins(logConsole);
    
    // Собираем функцию и запускаем её
    const fn = new AsyncFn(...Object.keys(builtinsObj), jsCode);
    await fn(...Object.values(builtinsObj));
    
    logConsole('✅ Выполнено успешно', 'ok');
    statusEl.textContent = 'Успешно ✓';
    statusEl.style.color = 'var(--accent3)';
  } catch (err) {
    logConsole('❌ ОШИБКА ВЫПОЛНЕНИЯ: ' + err.message, 'err');
    statusEl.textContent = 'Ошибка';
    statusEl.style.color = 'var(--danger)';
  }
}

        document.getElementById('pyBtn').click();

        updateHighlight();

      }

/* ================== ЗАГРУЗКА КОДА ИЗ ССЫЛКИ ================== */
window.addEventListener('load', () => {
  const urlParams = new URLSearchParams(window.location.search);
  const sharedCode = urlParams.get('code');
  
  if (sharedCode) {
    try {
      // Расшифровываем Base64 код из ссылки
      editor.value = decodeURIComponent(escape(atob(sharedCode)));
      updateHighlight();
      logConsole('✅ Код загружен из ссылки!', 'ok');
    } catch (e) {
      logConsole('❌ Ошибка чтения кода из ссылки.', 'err');
    }
  } else {
    // Стартовый код для новых пользователей
    editor.value = `// Добро пожаловать в BC+ IDE v3.0!\n// Справа тебе поможет локальный бот.\n\nсказать "Привет, разработчик!"\nпусть имя = "друг"\nсказать "Привет, " + имя + "!"\n\n// Попробуй команду ИИ:\nии "Напиши короткое приветствие на русском"`;
  }
  updateHighlight();
});

   ============================================================ */
async function runCode() {
  clearConsole(); 
  errorLines = []; 
  updateHighlight(); 
  
  // Закрываем игровое окно перед новым запуском
  const gameModal = document.getElementById('gameModal');
  if (gameModal) gameModal.classList.remove('open');
  
  const code = editor.value.trim();
  if (!code) {
    logConsole('Редактор пуст. Нечего запускать.', 'info');
    return;
  }

  let jsCode;
  try {
    // Переводим BC+ на JavaScript
    jsCode = bcToJs(code); 
  } catch (err) {
    // Если парсер нашел ошибку (например, незакрытая скобка)
    logConsole('❌ ОШИБКА ПЕРЕВОДА', 'err');
    err.message.split('\n').forEach(l => logConsole(l, 'err'));
    
    // Подсвечиваем строку с ошибкой
    const m = err.message.match(/Строка (\d+):/); 
    if (m) { 
      errorLines.push(parseInt(m[1])); 
      updateHighlight(); 
    }
    return;
  }

  logConsole('▶ Запуск main.bc+...', 'info');
  
  try {
    // Создаем асинхронную функцию (чтобы работал await для ИИ и ввода)
    const AsyncFn = Object.getPrototypeOf(async function(){}).constructor;
    
    // Получаем доступ к встроенным функциям (print, ввод, спросить_ии и т.д.)
    const b = buildBuiltins(logConsole);
    
    // Собираем функцию, передавая ей все встроенные команды как аргументы
    const fn = new AsyncFn(...Object.keys(b), jsCode);
    
    // Запускаем!
    await fn(...Object.values(b));
    logConsole('✅ Выполнено успешно', 'ok');
    
  } catch (err) { 
    // Ловим ошибки времени выполнения (например, деление на ноль)
    logConsole('❌ ОШИБКА ВЫПОЛНЕНИЯ: ' + err.message, 'err'); 
  }
}

/* ============================================================
   14. ПЕРЕВОД В PYTHON
   ============================================================ */
function translatePY() {
  const lines = editor.value.trim().split('\n');
  let out = [], indent = 0;
  const pad = () => '    '.repeat(indent); // В Python 4 пробела для отступа

  for (let raw of lines) {
    let line = raw.trim();
    if (!line) continue;

    // Комментарии
    if (line.startsWith('//')) { 
      out.push(pad() + '# ' + line.slice(2).trim()); 
      continue; 
    }

    // Закрытие блока
    if (line === '}') { 
      indent = Math.max(0, indent - 1); 
      continue; 
    }
    
    // Иначе
    if (line === 'иначе {' || line === 'иначе{') { 
      indent = Math.max(0, indent - 1); 
      out.push(pad() + 'else:'); 
      indent++; 
      continue; 
    }

    // Условия и циклы
    if (line.startsWith('если ') && line.endsWith('{')) { 
      let c = line.slice(5, -1).replace(/\s+то\s*$/, '').replace(/\bи\b/g,'and').replace(/\bили\b/g,'or').replace(/\bне\b/g,'not'); 
      out.push(pad() + 'if ' + c + ':'); 
      indent++; continue; 
    }
    if (line.startsWith('пока ') && line.endsWith('{')) { 
      let c = line.slice(5, -1).replace(/\s+то\s*$/, '').replace(/\bи\b/g,'and').replace(/\bили\b/g,'or').replace(/\bне\b/g,'not'); 
      out.push(pad() + 'while ' + c + ':'); 
      indent++; continue; 
    }
    if (line.startsWith('функция ') && line.endsWith('{')) { 
      out.push(pad() + 'def ' + line.slice(8, -1).trim() + ':'); 
      indent++; continue; 
    }

    // Простые команды
    if (line.startsWith('сказать ')) { out.push(pad() + 'print(' + line.slice(8).trim() + ')'); continue; }
    if (line.startsWith('пусть ')) { out.push(pad() + line.slice(6).trim()); continue; }
    if (line.startsWith('вернуть ')) { out.push(pad() + 'return ' + line.slice(8).trim()); continue; }
    
    // Специальные функции BC+
    if (line.includes(' = ввод ')) {
      const parts = line.split(' = ввод ');
      out.push(pad() + parts[0].replace(/^пусть\s+/, '') + ' = input(' + parts[1].trim() + ')');
      continue;
    }
    
    // Если не распознали, просто копируем
    out.push(pad() + line);
  }
  
  clearConsole(); 
  logConsole('🐍 Сгенерированный Python:', 'info'); 
  logConsole(out.join('\n') || '# Пустой код', 'ok');
}

   15. КНОПКА "ПОДЕЛИТЬСЯ" (ГЕНЕРАЦИЯ ССЫЛКИ)

function shareCode() {
  try {
    const encoded = btoa(unescape(encodeURIComponent(editor.value)));
    const url = window.location.origin + window.location.pathname + '?code=' + encoded;
    
    navigator.clipboard.writeText(url).then(() => {
      alert('🔗 Ссылка на код скопирована! Отправь её другу или сохрани в закладки.');
    }).catch(() => {
      prompt('Скопируй ссылку вручную:', url);
    });
  } catch (e) { 
    alert('Ошибка при создании ссылки: ' + e.message); 
  }
}

function addMessage(sender, text, codeToInsert = null) {
  const msgDiv = document.createElement('div');
  msgDiv.className = 'msg ' + sender;
  
  // Выводим текст (поддерживаем переносы строк)
  msgDiv.innerHTML = text.replace(/\n/g, '
');

  // Если бот прислал код, добавляем его
  if (codeToInsert) {
    const pre = document.createElement('pre');
    const code = document.createElement('code');
    code.textContent = codeToInsert;
    pre.appendChild(code);
    msgDiv.appendChild(pre);

    // Создаем кнопку "Вставить в редактор"
    const insertBtn = document.createElement('button');
    insertBtn.textContent = '📋 Вставить код';
    insertBtn.style.cssText = 'margin-top:8px; padding:6px 12px; border-radius:6px; border:none; background:var(--accent); color:#000; font-weight:bold; cursor:pointer; font-size:0.8rem; transition:0.3s;';
    
    // При клике — заменяем код в редакторе
    insertBtn.onclick = () => {
      editor.value = codeToInsert;
      updateHighlight();
      editor.focus();
      logConsole('Код вставлен из чата!', 'info');
    };
    msgDiv.appendChild(insertBtn);
  }

  chatArea.appendChild(msgDiv);
  chatArea.scrollTop = chatArea.scrollHeight; // Прокрутка вниз
}

function processBotCommand(input) {
  const cmd = input.trim().toLowerCase();
  
  // Показываем сообщение пользователя
  addMessage('user', input);

  // Ищем команду в словаре
  if (BOT_COMMANDS[cmd]) {
    const response = BOT_COMMANDS[cmd];
    
    // Обработка команд, которые меняют интерфейс
    if (cmd === '/сброс') {
      editor.value = '';
      updateHighlight();
      addMessage('bot', 'Редактор успешно очищен!');
    } else if (cmd === '/очистить') {
      clearConsole();
      addMessage('bot', 'Отладчик успешно очищен!');
    } else {
      // Обычный ответ (с кодом или без)
      addMessage('bot', response.text, response.code);
    }
  } else {
    // Если команда не найдена
    addMessage('bot', 'Неизвестная команда. Напиши /help, чтобы увидеть список.');
  }
}

// Активация бота (запрос к Cloudflare Worker)
async function activateBot() {
  const token = document.getElementById('botToken').value.trim();
  const chatId = document.getElementById('botChatId').value.trim();
  const btn = document.querySelector('#botModal .btn-ai');

  if (!token) {
    alert('Введи токен Telegram-бота!');
    return;
  }

  const originalText = btn.textContent;
  btn.disabled = true;
  btn.textContent = '⏳ Проверка...';
  
  logConsole('⚡ Отправка запроса на сервер...', 'info');

  try {
    // 1. Проверяем токен через Worker
    const res = await fetch(PROXY_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ platform: 'telegram', action: 'verify', token })
    });
    
    const data = await res.json();

    if (!data.ok) {
      logConsole('❌ Ошибка: ' + (data.description || 'Неверный токен'), 'err');
      btn.disabled = false; btn.textContent = originalText;
      return;
    }

    logConsole('✅ Токен верный! Бот: @' + data.result.username, 'ok');

    // 2. Если указан chatId, отправляем тестовое сообщение
    if (chatId) {
      logConsole('📨 Отправка тестового сообщения...', 'info');
      await fetch(PROXY_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          platform: 'telegram', 
          token: token, 
          action: 'sendMessage', 
          payload: { chat_id: chatId, text: '🎉 Привет! Я бот на BC+!' } 
        })
      });
      logConsole('✅ Сообщение отправлено в чат ' + chatId, 'ok');
    }

    // Закрываем окно с задержкой
    setTimeout(() => {
      closeModal('botModal');
      btn.disabled = false; 
      btn.textContent = originalText;
    }, 1500);

  } catch (e) {
    logConsole('❌ Ошибка сети при обращении к Worker: ' + e.message, 'err');
    btn.disabled = false; 
    btn.textContent = originalText;
  }
}

    editor.value = editor.value.substring(0, s) + '  ' + editor.value.substring(editor.selectionEnd); 
    editor.selectionStart = editor.selectionEnd = s + 2; 
    updateHighlight(); 
  }
});
